# Задание 1. Заполнение двумерных массивов

## Задача 1.3. Заполнение матриц

Для выполнения задания был создан класс Matrix.

### Класс Matrix

Matrix хранит элементы в единственном поле long[,] _values. Конструктор Matrix(n, m) читает первую матрицу по столбцам; Matrix(n, random) заполняет вторую случайно; Matrix(n) строит третью по рисунку. Ещё один конструктор копирует готовый массив. Размеры и элементы доступны через Rows, Columns и индексатор; ToString формирует таблицу.

Во всех трёх матрицах используется одно n. При случайном заполнении условие row + column < n − 1 выделяет область выше побочной диагонали. Третья матрица заполняется числами 1, 2, 3… по диагоналям от правого нижнего угла. На каждой диагонали движение идёт вверх и вправо; остальные элементы остаются нулевыми.

**Полный код Matrix для заданий 1–3**

```csharp
using System.Globalization;
using System.Text;
using CSharpLabs.Common;

namespace CSharpLabs.Lab3.Common;

public class Matrix
{
    private readonly long[,] _values;

    /// <summary>Вводит прямоугольную матрицу по столбцам сверху вниз.</summary>
    public Matrix(int rows, int columns)
    {
        ValidateDimensions(rows, columns);
        _values = new long[rows, columns];

        for (int column = 0; column < columns; column++)
        {
            for (int row = 0; row < rows; row++)
            {
                _values[row, column] = ConsoleInput.ReadInt(
                    $"Элемент [{row + 1}, {column + 1}]: ", int.MinValue, int.MaxValue);
            }
        }
    }

    public Matrix(int size, Random random)
    {
        ValidateDimensions(size, size);
        ArgumentNullException.ThrowIfNull(random);
        _values = new long[size, size];

        for (int row = 0; row < size; row++)
        {
            for (int column = 0; column < size; column++)
            {
                if (row + column < size - 1)
                {
                    _values[row, column] = random.Next(-12, 4566);
                }
                else
                {
                    _values[row, column] = random.Next(-1024, 1025);
                }
            }
        }
    }

    /// <summary>Заполняет нижний правый треугольник по диагоналям от правого нижнего угла.</summary>
    public Matrix(int size)
    {
        ValidateDimensions(size, size);
        _values = new long[size, size];
        long value = 1;

        for (int diagonal = 0; diagonal < size; diagonal++)
        {
            for (int offset = 0; offset <= diagonal; offset++)
            {
                _values[size - 1 - offset, size - 1 - diagonal + offset] = value;
                value++;
            }
        }
    }

    /// <summary>Создаёт независимую копию переданного массива.</summary>
    public Matrix(long[,] values)
    {
        ArgumentNullException.ThrowIfNull(values);
        ValidateDimensions(values.GetLength(0), values.GetLength(1));
        _values = (long[,])values.Clone();
    }

    public int Rows
    {
        get
        {
            return _values.GetLength(0);
        }
    }

    public int Columns
    {
        get
        {
            return _values.GetLength(1);
        }
    }

    public long this[int row, int column]
    {
        get
        {
            return _values[row, column];
        }
    }

    /// <summary>Находит первый при обходе по строкам блок 3×3 с максимальной суммой.</summary>
    public Matrix FindMaximumBlock(out int top, out int left, out long sum)
    {
        if (Rows < 3 || Columns < 3)
        {
            throw new InvalidOperationException("Для блока 3×3 нужны минимум три строки и три столбца.");
        }

        top = 0;
        left = 0;
        sum = long.MinValue;
        long current = 0;

        for (int row = 0; row <= Rows - 3; row++)
        {
            for (int column = 0; column <= Columns - 3; column++)
            {
                current = 0;

                for (int i = 0; i < 3; i++)
                {
                    for (int j = 0; j < 3; j++)
                    {
                        current = checked(current + _values[row + i, column + j]);
                    }
                }

                if (current > sum)
                {
                    sum = current;
                    top = row;
                    left = column;
                }
            }
        }

        long[,] block = new long[3, 3];

        for (int row = 0; row < 3; row++)
        {
            for (int column = 0; column < 3; column++)
            {
                block[row, column] = _values[top + row, left + column];
            }
        }

        return new Matrix(block);
    }

    public Matrix Transpose()
    {
        long[,] result = new long[Columns, Rows];

        for (int row = 0; row < Rows; row++)
        {
            for (int column = 0; column < Columns; column++)
            {
                result[column, row] = _values[row, column];
            }
        }

        return new Matrix(result);
    }

    public static Matrix operator +(Matrix first, Matrix second)
    {
        ArgumentNullException.ThrowIfNull(first);
        ArgumentNullException.ThrowIfNull(second);

        if (first.Rows != second.Rows || first.Columns != second.Columns)
        {
            throw new ArgumentException("Размеры слагаемых матриц должны совпадать.");
        }

        long[,] result = new long[first.Rows, first.Columns];

        for (int row = 0; row < first.Rows; row++)
        {
            for (int column = 0; column < first.Columns; column++)
            {
                result[row, column] = checked(first[row, column] + second[row, column]);
            }
        }

        return new Matrix(result);
    }

    public static Matrix operator *(long factor, Matrix matrix)
    {
        ArgumentNullException.ThrowIfNull(matrix);
        long[,] result = new long[matrix.Rows, matrix.Columns];

        for (int row = 0; row < matrix.Rows; row++)
        {
            for (int column = 0; column < matrix.Columns; column++)
            {
                result[row, column] = checked(factor * matrix[row, column]);
            }
        }

        return new Matrix(result);
    }

    public static Matrix operator *(Matrix matrix, long factor)
    {
        return factor * matrix;
    }

    public override string ToString()
    {
        int width = 1;

        foreach (long value in _values)
        {
            width = Math.Max(width, value.ToString(CultureInfo.InvariantCulture).Length);
        }

        StringBuilder result = new StringBuilder();
        string formattedValue = string.Empty;

        for (int row = 0; row < Rows; row++)
        {
            for (int column = 0; column < Columns; column++)
            {
                if (column > 0)
                {
                    result.Append(' ');
                }

                formattedValue = _values[row, column].ToString(CultureInfo.InvariantCulture);
                result.Append(formattedValue.PadLeft(width));
            }

            if (row < Rows - 1)
            {
                result.AppendLine();
            }
        }

        return result.ToString();
    }

    private static void ValidateDimensions(int rows, int columns)
    {
        ArgumentOutOfRangeException.ThrowIfNegativeOrZero(rows);
        ArgumentOutOfRangeException.ThrowIfNegativeOrZero(columns);
    }
}
```

### Условие

Первый массив n×m заполняется с клавиатуры по столбцам сверху вниз. Во втором массиве n×n выше побочной диагонали находятся случайные целые числа [-12; 4565], на диагонали и ниже — [-1024; 1024]. Третий массив заполняется по рисунку для произвольного n. Во всех трёх матрицах используется одно и то же n. Третья матрица заполняется по заданной закономерности, а не случайно.

### Описание задачи

Порядок выполнения примера:

1. В обычном запуске первая матрица n×m вводится с клавиатуры по столбцам сверху вниз. Для иллюстрации в демонстрационном режиме подготовлена матрица 5×3 с числами от 1 до 15.
2. Вторая матрица n×n создаётся случайно в заданных диапазонах; n совпадает с числом строк первой матрицы.
3. Третья матрица n×n заполняется по закономерности рисунка, без случайных чисел. При n=5 получается рисунок из условия; все три матрицы выводятся через ToString.

# Задание 2. Работа с двумерными массивами

## Задача 2.3. Подмассив с максимальной суммой

Для выполнения задания в класс Matrix был добавлен метод FindMaximumBlock.

### Метод поиска в классе Matrix

FindMaximumBlock перебирает позиции левого верхнего угла: от 0 до Rows − 3 и от 0 до Columns − 3 включительно. Для каждой позиции складываются девять элементов. Блоки могут перекрываться, а размеры массива не обязаны делиться на три.

Метод сохраняет наибольшую сумму и координаты блока. Начальное значение long.MinValue позволяет работать с отрицательными числами. При равных суммах выбирается первый блок при обходе по строкам. На экран координаты выводятся с единицы.

Полный код класса Matrix приведён в задаче 1.3.

### Условие

В массиве n×m найти подмассив 3×3 с максимальной суммой. Размеры могут быть не кратны трём.

1. Создана исходная матрица 4×5 с числами от 1 до 20.
2. Метод FindMaximumBlock проверил все возможные положения блока 3×3.
3. Выведены найденный блок, его сумма 126 и начало: строка 2, столбец 3.

## Задача 3.3. Матричное выражение

Для выполнения задания в класс Matrix были добавлены транспонирование и перегруженные операторы.

### Операции класса Matrix

Transpose меняет строки и столбцы местами. Операторы + и * перегружены для сложения матриц и умножения матрицы на число. Поэтому выражение записано как a.Transpose() + 2 * b + c.Transpose(). Приоритет остаётся обычным: умножение выполняется раньше сложения.

Матрицы A и C должны иметь одинаковые размеры, а B — размеры их транспонированных матриц. Квадратность не требуется. Операции создают новые матрицы; исходные не изменяются. checked обнаруживает переполнение long.

# Задание 4. Бинарные файлы

## Задача 4.3. Фильтрация чисел

Для выполнения задания был создан статический класс FileTasks.

### Статический класс FileTasks

FillBinaryNumbers создаёт исходный файл со случайными целыми числами от −100 до 100. FilterBinaryNumbers читает их по очереди и сохраняет числа, для которых value % m == 0 и value % n != 0. Порядок и повторения сохраняются.

Делители m и n не могут быть нулевыми. Число 0 не попадает в результат, поскольку делится на оба делителя. Приведение value к long предотвращает переполнение для int.MinValue % −1. Входной бинарный файл должен состоять из целых записей Int32.

**Методы FileTasks**

```csharp
public static void FillBinaryNumbers(string path, int count, Random random)
{
    ArgumentOutOfRangeException.ThrowIfNegative(count);
    ArgumentNullException.ThrowIfNull(random);
    using BinaryWriter writer = new BinaryWriter(File.Create(path));

    for (int i = 0; i < count; i++)
    {
        writer.Write(random.Next(-100, 101));
    }
}

public static void FilterBinaryNumbers(string source, string destination, int m, int n)
{
    ValidateDifferentPaths(source, destination);

    if (m == 0 || n == 0)
    {
        throw new ArgumentException("Делители m и n не должны быть равны нулю.");
    }

    using BinaryReader reader = new BinaryReader(File.OpenRead(source));

    if (reader.BaseStream.Length % sizeof(int) != 0)
    {
        throw new InvalidDataException("Бинарный файл должен содержать целые Int32.");
    }

    using BinaryWriter writer = new BinaryWriter(File.Create(destination));
    int value = 0;

    while (reader.BaseStream.Position < reader.BaseStream.Length)
    {
        value = reader.ReadInt32();

        // Расширение до long исключает переполнение int.MinValue % -1.
        if ((long)value % m == 0 && (long)value % n != 0)
        {
            writer.Write(value);
        }
    }
}

public static string ReadBinaryNumbers(string path)
{
    using BinaryReader reader = new BinaryReader(File.OpenRead(path));

    if (reader.BaseStream.Length % sizeof(int) != 0)
    {
        throw new InvalidDataException("Бинарный файл должен содержать целые Int32.");
    }

    StringBuilder result = new StringBuilder();

    while (reader.BaseStream.Position < reader.BaseStream.Length)
    {
        if (result.Length > 0)
        {
            result.Append(' ');
        }

        result.Append(reader.ReadInt32());
    }

    if (result.Length == 0)
    {
        return "(пусто)";
    }
    else
    {
        return result.ToString();
    }
}

private static void ValidateDifferentPaths(string source, string destination)
{
    if (string.Equals(Path.GetFullPath(source), Path.GetFullPath(destination),
        StringComparison.OrdinalIgnoreCase))
    {
        throw new ArgumentException("Исходный и выходной файлы должны различаться.");
    }
}
```

public static void FillPassengers(string path, Passenger[] passengers)
{
    ArgumentNullException.ThrowIfNull(passengers);
    XmlSerializer serializer = new XmlSerializer(typeof(Passenger[]));
    using MemoryStream xml = new MemoryStream();
    serializer.Serialize(xml, passengers);
    byte[] data = xml.ToArray();
    using BinaryWriter writer = new BinaryWriter(File.Create(path));
    writer.Write(data.Length);
    writer.Write(data);
}

public static Passenger[] ReadPassengers(string path)
{
    using BinaryReader reader = new BinaryReader(File.OpenRead(path));

    if (reader.BaseStream.Length < sizeof(int))
    {
        throw new InvalidDataException("В файле отсутствует заголовок длины XML.");
    }

    int length = reader.ReadInt32();

    if (length < 0 || length != reader.BaseStream.Length - sizeof(int))
    {
        throw new InvalidDataException("Повреждена длина XML в бинарном файле.");
    }

    using MemoryStream xml = new MemoryStream(reader.ReadBytes(length));
    XmlSerializer serializer = new XmlSerializer(typeof(Passenger[]));
    return (Passenger[])(serializer.Deserialize(xml)
        ?? throw new InvalidDataException("В файле нет массива пассажиров."));
}

            return true;
        }
    }

    return false;
}

# Задание 6. Текстовые файлы: одно число в строке

## Задача 6.3. Разность максимума и минимума

Для выполнения задания в класс FileTasks были добавлены методы заполнения и обработки текстового файла.

### Методы FileTasks для чисел по одному в строке

FillTextNumbers через StreamWriter с perLine=1 записывает случайные целые числа по одному в строке. FindRange читает файл построчно через StreamReader, находит минимум и максимум, затем возвращает максимум минус минимум.

Для одного числа результат равен нулю. Разность вычисляется в long, поэтому крайние значения int не вызывают переполнения. Пустой файл отклоняется: у него нельзя определить минимум и максимум.

Для показа содержимого используется ReadText: он открывает StreamReader и вызывает ReadToEnd. Все текстовые потоки работают с UTF-8 и закрываются через using.

**Методы FileTasks**

```csharp
public static void FillTextNumbers(string path, int count, int perLine, Random random)
{
    ArgumentOutOfRangeException.ThrowIfNegative(count);
    ArgumentOutOfRangeException.ThrowIfNegativeOrZero(perLine);
    ArgumentNullException.ThrowIfNull(random);
    using StreamWriter writer = new StreamWriter(path, false, Encoding.UTF8);

    for (int i = 0; i < count; i++)
    {
        writer.Write(random.Next(-100, 101).ToString(CultureInfo.InvariantCulture));

        if ((i + 1) % perLine == 0 || i == count - 1)
        {
            writer.WriteLine();
        }
        else
        {
            writer.Write(' ');
        }
    }
}

public static string ReadText(string path)
{
    using StreamReader reader = new StreamReader(path, Encoding.UTF8);
    return reader.ReadToEnd();
}

public static long FindRange(string path)
{
    using StreamReader reader = new StreamReader(path, Encoding.UTF8);
    string? line = reader.ReadLine();

    if (line is null)
    {
        throw new InvalidDataException("В файле нет чисел.");
    }

    int minimum = int.Parse(line, CultureInfo.InvariantCulture);
    int maximum = minimum;
    int value = 0;

    while ((line = reader.ReadLine()) is not null)
    {
        value = int.Parse(line, CultureInfo.InvariantCulture);
        minimum = Math.Min(minimum, value);
        maximum = Math.Max(maximum, value);
    }

    return (long)maximum - minimum;
}
```

### Методы FileTasks для нескольких чисел в строке

Пробелы, табуляция и пустые строки не мешают чтению. Если чисел нет, выдаётся ошибка. В обычном запуске задаются не меньше двух чисел и от двух до десяти чисел в полной строке, но не больше общего количества.

**Методы FileTasks**

```csharp
public static int FindMinimum(string path)
{
    using StreamReader reader = new StreamReader(path, Encoding.UTF8);
    int minimum = int.MaxValue;
    bool found = false;
    string[] tokens = Array.Empty<string>();
    string? line = null;

    while ((line = reader.ReadLine()) is not null)
    {
        tokens = line.Split((char[]?)null, StringSplitOptions.RemoveEmptyEntries);

        foreach (string token in tokens)
        {
            minimum = Math.Min(minimum, int.Parse(token, CultureInfo.InvariantCulture));
            found = true;
        }
    }

    if (!found)
    {
        throw new InvalidDataException("В файле нет чисел.");
    }

    return minimum;
}
```

### Метод отбора строк в FileTasks

FillTextLines записывает исходные строки через StreamWriter. FilterLines читает строки через StreamReader и записывает строку через StreamWriter, если она непустая и её первый символ совпадает с заданным. Регистр учитывается, начальные пробелы не удаляются.

Порядок строк сохраняется. Если совпадений нет, выходной файл будет пустым. Исходный и выходной пути должны различаться, чтобы не перезаписать исходный текст.

**Методы FileTasks**

```csharp
public static void FillTextLines(string path, string[] lines)
{
    ArgumentNullException.ThrowIfNull(lines);
    using StreamWriter writer = new StreamWriter(path, false, Encoding.UTF8);

    foreach (string line in lines)
    {
        writer.WriteLine(line);
    }
}

public static void FilterLines(string source, string destination, char firstCharacter)
{
    ValidateDifferentPaths(source, destination);
    using StreamReader reader = new StreamReader(source, Encoding.UTF8);
    using StreamWriter writer = new StreamWriter(destination, false, Encoding.UTF8);
    string? line = null;

    while ((line = reader.ReadLine()) is not null)
    {
        if (line.Length > 0 && line[0] == firstCharacter)
        {
            writer.WriteLine(line);
        }
    }
}
```

# Ручной ввод данных

Lab3Input читает параметры заданий через общий класс ConsoleInput из CSharpLabs/Common. Если A и C имеют размер n×m, для B задаётся размер m×n. Масса предмета должна быть положительной; порог m может быть любым числом. Делители не могут быть нулевыми. Пустые строки и начальные пробелы текста сохраняются. Элементы матриц читает конструктор Matrix(n, m), заполняя массив по столбцам.

**Полный код Lab3Input**

```csharp
using System.Globalization;
using CSharpLabs.Common;
using CSharpLabs.Lab3.Tasks;

namespace CSharpLabs.Lab3.Common;

internal static class Lab3Input
{
    public static void Run(int task)
    {
        switch (task)
        {
            case 1:
                Task1.Run();
                break;
            case 2:
                Task2.Run(ReadMatrix("Исходная матрица", 3));
                break;
            case 3:
                ReadExpression();
                break;
            case 4:
                int count = ConsoleInput.ReadInt("Количество чисел (0–100): ", 0, 100);
                int m = ReadNonzero("Делитель m: ");
                int n = ReadNonzero("Делитель n: ");
                Task4.Run(count, m, n);
                break;
            case 5:
                Passenger[] passengers = ReadPassengers();
                decimal mass = ReadMass("Порог m, кг: ", false);
                Task5.Run(passengers, mass);
                break;
            case 6:
            case 7:
                int minimumCount = 0;

                if (task == 6)
                {
                    minimumCount = 1;
                }
                else
                {
                    minimumCount = 2;
                }
                int numberCount = ConsoleInput.ReadInt($"Количество чисел ({minimumCount}–100): ", minimumCount, 100);
                int maximumPerLine = Math.Min(10, numberCount);
                int perLine = 0;

                if (task == 6)
                {
                    perLine = 1;
                }
                else
                {
                    perLine = ConsoleInput.ReadInt($"Чисел в строке (2–{maximumPerLine}): ", 2, maximumPerLine);
                }

                if (task == 6)
                {
                    Task6.Run(numberCount);
                }
                else
                {
                    Task7.Run(numberCount, perLine);
                }
                break;
            case 8:
                char firstCharacter = ReadCharacter();
                string[] lines = ReadLines();
                Task8.Run(lines, firstCharacter);
                break;
        }
    }

    private static void ReadExpression()
    {
        int rows = ConsoleInput.ReadInt("Число строк A и C (1–20): ", 1, 20);
        int columns = ConsoleInput.ReadInt("Число столбцов A и C (1–20): ", 1, 20);
        Console.WriteLine("Матрица A, ввод по столбцам:");
        Matrix a = new Matrix(rows, columns);
        Console.WriteLine("Матрица B, размеры обратные размерам A:");
        Matrix b = new Matrix(columns, rows);
        Console.WriteLine("Матрица C:");
        Matrix c = new Matrix(rows, columns);
        Task3.Run(a, b, c);
    }

    private static Passenger[] ReadPassengers()
    {
        int count = ConsoleInput.ReadInt("Количество пассажиров (0–20): ", 0, 20);
        Passenger[] passengers = new Passenger[count];
        string name = string.Empty;
        int itemsCount = 0;
        BaggageItem[] items = Array.Empty<BaggageItem>();
        string itemName = string.Empty;

        for (int i = 0; i < count; i++)
        {
            name = ConsoleInput.ReadRequiredText($"Имя пассажира {i + 1}: ");
            itemsCount = ConsoleInput.ReadInt("Количество единиц багажа (0–20): ", 0, 20);
            items = new BaggageItem[itemsCount];

            for (int j = 0; j < itemsCount; j++)
            {
                itemName = ConsoleInput.ReadRequiredText($"Название единицы {j + 1}: ");
                items[j] = new BaggageItem(itemName, ReadMass("Масса, кг: "));
            }

            passengers[i] = new Passenger(name, items);
        }

        return passengers;
    }

    private static char ReadCharacter()
    {
        string symbol = string.Empty;

        do
        {
            Console.Write("Начальный символ (ровно один, пробел допустим): ");
            symbol = Console.ReadLine() ?? throw new EndOfStreamException();

            if (symbol.Length != 1)
            {
                Console.WriteLine("Введите ровно один символ.");
            }
        }
        while (symbol.Length != 1);

        return symbol[0];
    }

    private static string[] ReadLines()
    {
        int count = ConsoleInput.ReadInt("Количество строк (0–30): ", 0, 30);
        string[] lines = new string[count];

        for (int i = 0; i < count; i++)
        {
            Console.Write($"Строка {i + 1}: ");
            lines[i] = Console.ReadLine() ?? throw new EndOfStreamException();
        }

        return lines;
    }

    private static Matrix ReadMatrix(string title, int minimum = 1)
    {
        Console.WriteLine(title);
        int rows = ConsoleInput.ReadInt($"Число строк ({minimum}–20): ", minimum, 20);
        int columns = ConsoleInput.ReadInt($"Число столбцов ({minimum}–20): ", minimum, 20);
        return new Matrix(rows, columns);
    }

    private static int ReadNonzero(string prompt)
    {
        int value = 0;

        while (true)
        {
            value = ConsoleInput.ReadInt(prompt, int.MinValue, int.MaxValue);

            if (value != 0)
            {
                return value;
            }

            Console.WriteLine("Делитель не должен быть равен нулю.");
        }
    }

    private static decimal ReadMass(string prompt, bool requirePositive = true)
    {
        string text = string.Empty;
        decimal value = 0;

        while (true)
        {
            text = ConsoleInput.ReadRequiredText(prompt).Replace(',', '.');

            if (decimal.TryParse(text, NumberStyles.AllowLeadingSign | NumberStyles.AllowDecimalPoint,
                CultureInfo.InvariantCulture, out value) && (!requirePositive || value > 0))
            {
                return value;
            }

            if (requirePositive)
            {
                Console.WriteLine("Введите положительную массу; дробную часть отделите точкой или запятой.");
            }
            else
            {
                Console.WriteLine("Введите число; дробную часть отделите точкой или запятой.");
            }
        }
    }
}
```
