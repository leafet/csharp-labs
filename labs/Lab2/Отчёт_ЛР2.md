# Задание 1. Создание сущностей

## Задача 1.3. Имена

Для выполнения задания был создан класс PersonFullName.

### Класс PersonFullName

Класс хранит личное имя, фамилию и отчество в трёх полях. Любая отдельная часть может отсутствовать, но хотя бы одна должна быть задана. Свойства Name, SecondName и Patronymic позволяют читать и изменять части ФИО.

Конструктор с тремя параметрами нормализует значения и проверяет общее состояние. Сеттеры применяют ту же проверку перед присваиванием, поэтому попытка удалить последнюю часть оставляет прежнее значение. ToString собирает заданные части в порядке «фамилия имя отчество» без лишних пробелов. Ниже приведён полный код класса; его перегрузки конструкторов используются также в задаче 4.5.

**Полный код PersonFullName**

```csharp
public class PersonFullName
{
    private string? _name;
    private string? _secondName;
    private string? _patronymic;

    public PersonFullName(string? name, string? secondName, string? patronymic)
    {
        _name = OptionalText(name);
        _secondName = OptionalText(secondName);
        _patronymic = OptionalText(patronymic);
        ValidateParts(_name, _secondName, _patronymic);
    }

    public PersonFullName(string? name, string? secondName) : this(name, secondName, null)
    {
    }

    public PersonFullName(string? name) : this(name, null, null)
    {
    }

    public string? Name
    {
        get
        {
            return _name;
        }
        set
        {
            string? text = OptionalText(value);
            ValidateParts(text, _secondName, _patronymic);

            _name = text;
        }
    }

    public string? SecondName
    {
        get
        {
            return _secondName;
        }
        set
        {
            string? text = OptionalText(value);
            ValidateParts(_name, text, _patronymic);

            _secondName = text;
        }
    }

    public string? Patronymic
    {
        get
        {
            return _patronymic;
        }
        set
        {
            string? text = OptionalText(value);
            ValidateParts(_name, _secondName, text);

            _patronymic = text;
        }
    }

    public override string ToString()
    {
        string result = _secondName ?? string.Empty;

        if (_name is not null)
        {
            result += result.Length > 0 ? " " + _name : _name;
        }

        if (_patronymic is not null)
        {
            result += result.Length > 0 ? " " + _patronymic : _patronymic;
        }

        return result;
    }

    private static void ValidateParts(string? name, string? secondName, string? patronymic)
    {
        if (name is null && secondName is null && patronymic is null)
        {
            throw new ArgumentException("Должна быть задана хотя бы одна часть ФИО.");
        }
    }

    private static string? OptionalText(string? value)
    {
        string? result = value?.Trim();
        return string.IsNullOrEmpty(result) ? null : result;
    }
}
```

## Задача 2.2. Человек с именем

Для выполнения задания был модифицирован класс Person.

### Класс Person

Person хранит ссылку на PersonFullName и целочисленный рост. Свойство FullName не принимает null; Height требует положительного значения. Объект ФИО не копируется: изменение его через другую ссылку видно и у человека.

ToString выводит ФИО и рост. Ниже показана итоговая версия класса целиком. Поле Father и связанные с ним методы объясняются в задаче 2.3, а перегрузки конструкторов — в задаче 4.6.

**Полный код Person**

```csharp
public class Person
{
    private PersonFullName _fullName;
    private int _height;
    private Person? _father;

    public Person(PersonFullName fullName, int height, Person? father)
    {
        _fullName = fullName ?? throw new ArgumentNullException(nameof(fullName));
        Height = height;
        Father = father;
        CompleteNameFromFather();
    }

    public Person(string? name, int height, Person father)
        : this(new PersonFullName(name), height, father)
    {
    }

    public Person(PersonFullName fullName, int height) : this(fullName, height, null)
    {
    }

    public Person(string? name, int height) : this(new PersonFullName(name), height, null)
    {
    }

    public PersonFullName FullName
    {
        get
        {
            return _fullName;
        }
        set
        {
            _fullName = value ?? throw new ArgumentNullException(nameof(value));
        }
    }

    public int Height
    {
        get
        {
            return _height;
        }
        set
        {
            if (value <= 0)
            {
                throw new ArgumentOutOfRangeException(nameof(value), "Рост должен быть положительным.");
            }

            _height = value;
        }
    }

    public Person? Father
    {
        get
        {
            return _father;
        }
        set
        {
            if (ReferenceEquals(this, value))
            {
                throw new ArgumentException("Человек не может быть собственным отцом.", nameof(value));
            }

            _father = value;
        }
    }

    public override string ToString()
    {
        CompleteNameFromFather();

        return $"{_fullName}, рост: {_height}";
    }

    private void CompleteNameFromFather()
    {
        if (_father is null)
        {
            return;
        }

        if (_fullName.SecondName is null)
        {
            _fullName.SecondName = _father.FullName.SecondName;
        }

        string? fatherName = _father.FullName.Name;

        if (_fullName.Patronymic is null && fatherName is not null)
        {
            _fullName.Patronymic = MakePatronymic(fatherName);
        }
    }

    private static string MakePatronymic(string name)
    {
        if (string.Equals(name, "Лев", StringComparison.OrdinalIgnoreCase))
        {
            return "Львович";
        }

        if (string.Equals(name, "Пётр", StringComparison.OrdinalIgnoreCase) ||
            string.Equals(name, "Петр", StringComparison.OrdinalIgnoreCase))
        {
            return "Петрович";
        }

        if (name.EndsWith('й'))
        {
            return name[..^1] + "евич";
        }

        return name + "ович";
    }
}
```

# Задание 3. Наборы объектов

## Задача 3.3. Города

Для выполнения задания были созданы классы City и Route.

### Классы City и Route

City хранит проверяемое название и список исходящих маршрутов. Конструктор создаёт пустой список, AddRoute запрещает путь в самого себя и повторный путь к тому же объекту назначения. Свойство Routes отдаёт список только для чтения. City.ToString выводит название и все исходящие пары «город:стоимость».

Route представляет одно направление: ссылку на город назначения и положительную стоимость. Назначение нельзя заменить после создания; стоимость допускает изменение только на положительное число. Если город переименовать, Route.ToString покажет новое название, поскольку хранится ссылка на City. Обратное направление создаётся отдельным маршрутом. Ниже приведены оба класса полностью.

**Полный код City**

```csharp
public class City
{
    private string _name;
    private readonly List<Route> _routes;

    public City(string name)
    {
        _name = ValidateName(name);
        _routes = new List<Route>();
    }

    public string Name
    {
        get
        {
            return _name;
        }
        set
        {
            _name = ValidateName(value);
        }
    }

    public IReadOnlyList<Route> Routes
    {
        get
        {
            return _routes.AsReadOnly();
        }
    }

    public void AddRoute(City destination, int cost)
    {
        if (destination is null)
        {
            throw new ArgumentNullException(nameof(destination));
        }

        if (ReferenceEquals(this, destination))
        {
            throw new ArgumentException("Путь должен вести в другой город.", nameof(destination));
        }

        for (int i = 0; i < _routes.Count; i++)
        {
            if (ReferenceEquals(_routes[i].Destination, destination))
            {
                throw new ArgumentException("Путь в этот город уже существует.", nameof(destination));
            }
        }

        _routes.Add(new Route(destination, cost));
    }

    public override string ToString()
    {
        StringBuilder result = new StringBuilder(_name);

        result.Append(" [");

        for (int i = 0; i < _routes.Count; i++)
        {
            if (i > 0)
            {
                result.Append(", ");
            }

            result.Append(_routes[i]);
        }

        result.Append(']');

        return result.ToString();
    }

    private static string ValidateName(string? value)
    {
        string? name = value?.Trim();

        if (string.IsNullOrEmpty(name))
        {
            throw new ArgumentException("Название города не может быть пустым.", nameof(value));
        }

        return name;
    }
}
```

**Полный код Route**

```csharp
public class Route
{
    private readonly City _destination;
    private int _cost;

    public Route(City destination, int cost)
    {
        _destination = destination ?? throw new ArgumentNullException(nameof(destination));
        Cost = cost;
    }

    public City Destination
    {
        get
        {
            return _destination;
        }
    }

    public int Cost
    {
        get
        {
            return _cost;
        }
        set
        {
            if (value <= 0)
            {
                throw new ArgumentOutOfRangeException(nameof(value), "Стоимость должна быть положительной.");
            }

            _cost = value;
        }
    }

    public override string ToString()
    {
        return $"{_destination.Name}:{_cost}";
    }
}
```

# Задание 5. Методы объектов

## Задача 5.2. Кот мяукает

Для выполнения задания был создан класс Cat.

### Класс Cat

Cat хранит непустое имя. Конструктор и свойство Name используют одну проверку: пробелы по краям удаляются, пустая строка отклоняется. ToString возвращает «кот: Имя».

Meow() вызывает Meow(1), а Meow(int count) печатает count повторений «мяу», разделённых дефисом. Класс принимает любое положительное количество; ноль и отрицательное число отклоняются. В меню ручного ввода установлен диапазон 1–100. Ниже приведён полный код класса.

**Полный код Cat**

```csharp
public class Cat
{
    private string _name;

    public Cat(string name)
    {
        _name = ValidateName(name);
    }

    public string Name
    {
        get
        {
            return _name;
        }
        set
        {
            _name = ValidateName(value);
        }
    }

    public void Meow()
    {
        Meow(1);
    }

    public void Meow(int count)
    {
        if (count < 1)
        {
            throw new ArgumentOutOfRangeException(nameof(count), "Количество мяуканий должно быть положительным.");
        }

        StringBuilder sound = new StringBuilder(_name + ": ");

        for (int i = 0; i < count; i++)
        {
            if (i > 0)
            {
                sound.Append('-');
            }

            sound.Append("мяу");
        }

        sound.Append('!');

        Console.WriteLine(sound);
    }

    public override string ToString()
    {
        return "кот: " + _name;
    }

    private static string ValidateName(string? value)
    {
        string? name = value?.Trim();

        if (string.IsNullOrEmpty(name))
        {
            throw new ArgumentException("Имя кота не может быть пустым.", nameof(value));
        }

        return name;
    }
}
```
